% Two-column entry heading.
% Each side lives in its own box, so a long title wraps inside its own
% column instead of running into the right-aligned date. The whole row
% is wrapped in a \makebox of exactly \textwidth, which keeps the right
% column flush right even though the document is \raggedright.
\newcommand{\heading}[2]{%
  \noindent\makebox[\textwidth][l]{%
    \hspace*{\entryIndent}%
    \begin{minipage}[t]{\dimexpr0.70\textwidth-\entryIndent\relax}%
      \raggedright\strut #1%
    \end{minipage}%
    \hfill
    \begin{minipage}[t]{0.28\textwidth}%
      \raggedleft\strut #2%
    \end{minipage}%
  }\par
}

\newcommand{\headingBf}[2]{
  \heading{\textbf{#1}}{\textbf{#2}}
}

\newcommand{\headingIt}[2]{
  \heading{\textit{#1}}{\textit{#2}}
}

% Two-column skills row, same boxed construction.

    \begin{minipage}[t]{0.40\textwidth}\raggedright\strut #2\end{minipage}%
  }\par\vspace{1.5pt}
}
